    'Garden Fate' => 400000,
    'Melody Lawn' => 600000,
    'Celestial Luxe' => 1000000

    'Basic' => 60000,
    'Premium' => 90000,
    'Luxury' => 120000
);

// Catering prices
$catering_prices = array(
    'Paneer Butter Masala' => 220,
    'Veg Biryani' => 200,
    'Chappati' => 15,
    'Veg Korma' => 180,
    'Chicken Biryani' => 350,

// Handle POST requests
if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    header('Content-Type: application/json');
    $action = isset($_POST['action']) ? $_POST['action'] : '';

    if ($action === 'get_yearly_bookings') {
        $current_year = date('Y');
        $table_check = $mysqli->query("SHOW TABLES LIKE 'bookings'");
        if ($table_check->num_rows === 0) {
            echo json_encode(array('success' => false, 'error' => 'Bookings table does not exist'));
            ob_end_flush();
            exit;
        }

        $stmt = $mysqli->prepare("SELECT MONTH(created_at) as month, COUNT(*) as count FROM bookings WHERE YEAR(created_at) = ? GROUP BY MONTH(created_at)");
        if ($stmt === false) {
            echo json_encode(array('success' => false, 'error' => 'Query preparation failed: ' . $mysqli->error));
            ob_end_flush();
            exit;
        }
        $stmt->bind_param("i", $current_year);
        if (!$stmt->execute()) {
            echo json_encode(array('success' => false, 'error' => 'Query execution failed: ' . $stmt->error));
            $stmt->close();
            ob_end_flush();
            exit;
        }
        $result = $stmt->get_result();
        $data = array_fill(1, 12, 0);
        while ($row = $result->fetch_assoc()) {
            $data[(int)$row['month']] = (int)$row['count'];
        }
        $dataPoints = array();
        $months = array('Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec');
        for ($i = 1; $i <= 12; $i++) {
            $dataPoints[] = array('label' => $months[$i-1], 'y' => $data[$i]);
        }
        echo json_encode(array('success' => true, 'dataPoints' => $dataPoints));
        $stmt->close();
        ob_end_flush();
        exit;
    } elseif ($action === 'get_monthly_bookings') {
        $current_year = date('Y');
        $current_month = date('m');
        $days_in_month = cal_days_in_month(CAL_GREGORIAN, $current_month, $current_year);
        $table_check = $mysqli->query("SHOW TABLES LIKE 'bookings'");
        if ($table_check->num_rows === 0) {
            echo json_encode(array('success' => false, 'error' => 'Bookings table does not exist'));
            ob_end_flush();
            exit;
        }

        $stmt = $mysqli->prepare("SELECT DAY(created_at) as day, COUNT(*) as count FROM bookings WHERE YEAR(created_at) = ? AND MONTH(created_at) = ? GROUP BY DAY(created_at)");
        if ($stmt === false) {
            echo json_encode(array('success' => false, 'error' => 'Query preparation failed: ' . $mysqli->error));
            ob_end_flush();
            exit;
        }
        $stmt->bind_param("ii", $current_year, $current_month);
        if (!$stmt->execute()) {
            echo json_encode(array('success' => false, 'error' => 'Query execution failed: ' . $stmt->error));
            $stmt->close();
            ob_end_flush();
            exit;
        }
        $result = $stmt->get_result();
        $data = array_fill(1, $days_in_month, 0);
        while ($row = $result->fetch_assoc()) {
            $data[(int)$row['day']] = (int)$row['count'];
        }
        $dataPoints = array();
        for ($i = 1; $i <= $days_in_month; $i++) {
            $dataPoints[] = array('label' => $i, 'y' => $data[$i]);
        }
        echo json_encode(array('success' => true, 'dataPoints' => $dataPoints));
        $stmt->close();
        ob_end_flush();
        exit;

        $current_year = date('Y');
        $table_check = $mysqli->query("SHOW TABLES LIKE 'bookings'");
        if ($table_check->num_rows === 0) {
            echo json_encode(array('success' => false, 'error' => 'Bookings table does not exist'));
            ob_end_flush();
            exit;

        $stmt = $mysqli->prepare("SELECT MONTH(created_at) as month, decoration, COUNT(*) as count FROM bookings WHERE YEAR(created_at) = ? AND hall_name IN ('Oasis Lawn', 'Celestial Luxe') AND decoration IN ('Premium', 'Luxury') GROUP BY MONTH(created_at), decoration");
        if ($stmt === false) {
            echo json_encode(array('success' => false, 'error' => 'Query preparation failed: ' . $mysqli->error));
            ob_end_flush();
            exit;
        }
        $stmt->bind_param("i", $current_year);
        if (!$stmt->execute()) {
            echo json_encode(array('success' => false, 'error' => 'Query execution failed: ' . $stmt->error));
            $stmt->close();
            ob_end_flush();
            exit;
        }

        $data_5 = array_fill(1, 12, 0);
        $data_10 = array_fill(1, 12, 0);

            $month = (int)$row['month'];
            if ($row['decoration'] === 'Premium') {
                $data_5[$month] = (int)$row['count'];
            } elseif ($row['decoration'] === 'Luxury') {
                $data_10[$month] = (int)$row['count'];
            }
        }
        $dataPoints_5 = array();
        $dataPoints_10 = array();
        $months = array('Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec');
        for ($i = 1; $i <= 12; $i++) {
            $dataPoints_5[] = array('label' => $months[$i-1], 'y' => $data_5[$i]);
            $dataPoints_10[] = array('label' => $months[$i-1], 'y' => $data_10[$i]);
        }
        echo json_encode(array(
            'success' => true,
            'dataPoints_5' => $dataPoints_5,
            'dataPoints_10' => $dataPoints_10
        ));
        $stmt->close();
        ob_end_flush();

            echo json_encode(array('success' => false, 'error' => 'Query preparation failed: ' . $mysqli->error));
            ob_end_flush();
            exit;
        }
        if (!$stmt->execute()) {
            echo json_encode(array('success' => false, 'error' => 'Query execution failed: ' . $stmt->error));
            $stmt->close();
            ob_end_flush();
            exit;
        }
        $result = $stmt->get_result();
        $bookings = $result->fetch_all(MYSQLI_ASSOC);
        $discount_bookings = array();
        foreach ($bookings as $booking) {
            $hall_price = isset($halls[$booking['hall_name']]) ? $halls[$booking['hall_name']] : 0;
            $decoration_price = isset($decoration_prices[$booking['decoration']]) ? $decoration_prices[$booking['decoration']] : 0;
            $catering_total = 0;
            if (!empty($booking['catering_item'])) {
                $catering_items = explode(', ', $booking['catering_item']);
                foreach ($catering_items as $item) {
                    $catering_total += isset($catering_prices[$item]) ? $catering_prices[$item] : 0;
                }
                $catering_total *= $booking['plates'];
            }
            $original_amount = $hall_price + $decoration_price + $catering_total;
            $discount_percent = 0;
            if ($booking['decoration'] === 'Premium') {
                $discount_percent = 5;
            } elseif ($booking['decoration'] === 'Luxury') {
                $discount_percent = 10;
            }
            $booking['discount_percent'] = $discount_percent;
            $booking['original_amount'] = $original_amount;
            $discount_bookings[] = $booking;
        }
        echo json_encode(array('success' => true, 'bookings' => $discount_bookings));
        $stmt->close();
        ob_end_flush();
        exit;

        $stmt = $mysqli->prepare("SELECT hall_name, booking_date, time_slot FROM bookings");
        if ($stmt === false) {
            echo json_encode(array('success' => false, 'error' => 'Query preparation failed: ' . $mysqli->error));
            ob_end_flush();
            exit;
        }
        if (!$stmt->execute()) {
            echo json_encode(array('success' => false, 'error' => 'Query execution failed: ' . $stmt->error));
            $stmt->close();
            ob_end_flush();
            exit;

        $bookings = $result->fetch_all(MYSQLI_ASSOC);

            $title = $booking['hall_name'] . ' (' . ($booking['time_slot'] === 'day' ? 'Day' : 'Evening') . ')';
            $events[] = array(
                'title' => $title,
                'start' => $booking['booking_date'],
                'time_slot' => $booking['time_slot'],
                'backgroundColor' => '#dc3545',
                'borderColor' => '#dc3545',
                'textColor' => '#ffffff',
                'extendedProps' => array(
                    'hall_name' => $booking['hall_name'],
                    'time_slot' => $booking['time_slot'] === 'day' ? 'Day Time' : 'Evening Time'
                )
            );
        }
        echo json_encode(array('success' => true, 'events' => $events));
        $stmt->close();
        ob_end_flush();
        exit;

        $booking_date = isset($_POST['booking_date']) ? $mysqli->real_escape_string($_POST['booking_date']) : '';
        $time_slot = isset($_POST['time_slot']) ? $mysqli->real_escape_string($_POST['time_slot']) : '';
        $decoration = isset($_POST['decoration']) ? $mysqli->real_escape_string($_POST['decoration']) : '';
        $total_amount = isset($_POST['total_amount']) ? (int)$_POST['total_amount'] : 0;
        $plates = isset($_POST['plates']) ? (int)$_POST['plates'] : 0;
        $catering_item = isset($_POST['catering_item']) ? $mysqli->real_escape_string($_POST['catering_item']) : '';

        $stmt_booking = $mysqli->prepare("SELECT name, email FROM bookings WHERE id = ?");
        if ($stmt_booking === false) {
            echo json_encode(array('success' => false, 'error' => 'Query preparation failed: ' . $mysqli->error));
            ob_end_flush();
            exit;
        }
        $stmt_booking->bind_param("i", $booking_id);
        if (!$stmt_booking->execute()) {
            echo json_encode(array('success' => false, 'error' => 'Query execution failed: ' . $stmt_booking->error));
            $stmt_booking->close();
            ob_end_flush();
            exit;
        }
        $stmt_booking->bind_result($original_name, $logged_in_email);
        $stmt_booking->fetch();
        $stmt_booking->close();

        $currentDate = date("Y-m-d");
        if ($booking_date <= $currentDate) {
            echo json_encode(array('success' => false, 'error' => 'Booking date must be greater than today.'));
            ob_end_flush();
            exit;
        }

        $stmt_select = $mysqli->prepare("SELECT * FROM bookings WHERE hall_name = ? AND booking_date = ? AND time_slot = ? AND id != ?");
        if ($stmt_select === false) {
            echo json_encode(array('success' => false, 'error' => 'Query preparation failed: ' . $mysqli->error));
            ob_end_flush();
            exit;
        }
        $stmt_select->bind_param("sssi", $hall_name, $booking_date, $time_slot, $booking_id);
        if (!$stmt_select->execute()) {
            echo json_encode(array('success' => false, 'error' => 'Query execution failed: ' . $stmt_select->error));
            $stmt_select->close();
            ob_end_flush();
            exit;
        }
        $result = $stmt_select->get_result();
        if ($result->num_rows > 0) {
            echo json_encode(array('success' => false, 'error' => 'This hall is already booked for this slot.'));
            $stmt_select->close();
            ob_end_flush();
            exit;
        }
        $stmt_select->close();

        $discount = 0;
        $eligibleHalls = array("Oasis Lawn", "Celestial Luxe");
        if (in_array($hall_name, $eligibleHalls)) {
            if ($decoration === "Premium") {
                $discount = 0.05;
            } elseif ($decoration === "Luxury") {
                $discount = 0.10;
            }
        }
        $finalAmount = $discount > 0 ? $total_amount - ($total_amount * $discount) : $total_amount;

        $stmt = $mysqli->prepare("UPDATE bookings SET name = ?, email = ?, contact = ?, hall_name = ?, booking_date = ?, time_slot = ?, catering_item = ?, decoration = ?, plates = ?, total_amount = ? WHERE id = ?");
        if ($stmt === false) {
            echo json_encode(array('success' => false, 'error' => 'Query preparation failed: ' . $mysqli->error));
            ob_end_flush();
            exit;
        }
        $stmt->bind_param("sssssssiii", $original_name, $logged_in_email, $contact, $hall_name, $booking_date, $time_slot, $catering_item, $decoration, $plates, $finalAmount, $booking_id);
        if ($stmt->execute()) {
            $discountText = $discount > 0 ? ($discount * 100) . '% discount applied. ' : '';
            echo json_encode(array(
                'success' => true,
                'message' => 'Booking Updated! 🎉 ' . $discountText . 'Final Amount: ₹' . $finalAmount,
                'booking' => array(
                    'id' => $booking_id,
                    'name' => $original_name,
                    'email' => $logged_in_email,
                    'contact' => $contact,
                    'hall_name' => $hall_name,
                    'booking_date' => $booking_date,
                    'time_slot' => $time_slot,
                    'decoration' => $decoration,
                    'plates' => $plates,
                    'total_amount' => $finalAmount,
                    'created_at' => date('Y-m-d H:i:s')
                )
            ));
        } else {
            echo json_encode(array('success' => false, 'error' => 'Query execution failed: ' . $stmt->error));
        }
        $stmt->close();
        ob_end_flush();
        exit;
    } elseif ($action === 'get_booking') {
        $id = isset($_POST['id']) ? (int)$_POST['id'] : 0;
        $stmt = $mysqli->prepare("SELECT id, contact, hall_name, booking_date, time_slot, plates, decoration, catering_item, created_at FROM bookings WHERE id = ?");
        if ($stmt === false) {
            echo json_encode(array('success' => false, 'error' => 'Query preparation failed: ' . $mysqli->error));
            ob_end_flush();
            exit;
        }
        $stmt->bind_param("i", $id);
        if (!$stmt->execute()) {
            echo json_encode(array('success' => false, 'error' => 'Query execution failed: ' . $stmt->error));
            $stmt->close();
            ob_end_flush();
            exit;
        }
        $result = $stmt->get_result();
        if ($booking = $result->fetch_assoc()) {
            echo json_encode(array('success' => true, 'booking' => $booking));
        } else {
            echo json_encode(array('success' => false, 'error' => 'Booking not found.'));
        }
        $stmt->close();
        ob_end_flush();
        exit;
    } elseif ($action === 'get_bookings_by_hall') {
        $hall_name = isset($_POST['hall_name']) ? $mysqli->real_escape_string($_POST['hall_name']) : '';
        $stmt = $mysqli->prepare("SELECT * FROM bookings WHERE hall_name = ?");
        if ($stmt === false) {
            echo json_encode(array('success' => false, 'error' => 'Query preparation failed: ' . $mysqli->error));
            ob_end_flush();
            exit;
        }
        $stmt->bind_param("s", $hall_name);
        if (!$stmt->execute()) {
            echo json_encode(array('success' => false, 'error' => 'Query execution failed: ' . $stmt->error));
            $stmt->close();
            ob_end_flush();
            exit;
        }
        $result = $stmt->get_result();
        $bookings = $result->fetch_all(MYSQLI_ASSOC);
        echo json_encode($bookings);
        $stmt->close();
        ob_end_flush();
        exit;
    } elseif ($action === 'delete_user') {
        $id = isset($_POST['id']) ? (int)$_POST['id'] : 0;
        $stmt = $mysqli->prepare("DELETE FROM users WHERE id = ?");
        if ($stmt === false) {
            echo json_encode(array('success' => false, 'error' => 'Query preparation failed: ' . $mysqli->error));
            ob_end_flush();
            exit;
        }
        $stmt->bind_param("i", $id);
        if ($stmt->execute()) {
            echo json_encode(array('success' => true));
        } else {
            echo json_encode(array('success' => false, 'error' => 'Query execution failed: ' . $stmt->error));
        }
        $stmt->close();
        ob_end_flush();
        exit;
    } elseif ($action === 'delete_booking') {
        $id = isset($_POST['id']) ? (int)$_POST['id'] : 0;
        $stmt = $mysqli->prepare("DELETE FROM bookings WHERE id = ?");
        if ($stmt === false) {
            echo json_encode(array('success' => false, 'error' => 'Query preparation failed: ' . $mysqli->error));
            ob_end_flush();
            exit;
        }
        $stmt->bind_param("i", $id);
        if ($stmt->execute()) {
            echo json_encode(array('success' => true));
        } else {
            echo json_encode(array('success' => false, 'error' => 'Query execution failed: ' . $stmt->error));
        }
        $stmt->close();
        ob_end_flush();
        exit;
    }
    ob_end_flush();
}

        // Sidebar open/close
        function openNav() {

            document.getElementById("main").style.marginLeft = "250px";
        }

        function closeNav() {
            document.getElementById("mySidenav").style.width = "0";

        }

        // Show/hide sections
        function showSection(sectionId) {
            var sections = document.getElementsByClassName('section');
            for (var i = 0; i < sections.length; i++) {
                sections[i].style.display = 'none';
            }
            document.getElementById(sectionId).style.display = 'block';
            closeNav();
            if (sectionId === 'dashboardSection') {
                loadCharts();
                loadCalendar();
            } else if (sectionId === 'discountsSection') {
                loadDiscounts();
            }
        }

        // Load CanvasJS charts
        function loadCharts() {
            $.ajax({
                url: 'admin_panel.php',
                type: 'POST',
                data: { action: 'get_yearly_bookings' },
                dataType: 'json',
                success: function(response) {
                    if (response.success) {
                        var chart = new CanvasJS.Chart("yearlyBookingsChart", {
                            animationEnabled: true,
                            theme: "light2",
                            title: { text: "Bookings by Month in <?php echo date('Y'); ?>" },
                            axisY: { title: "Number of Bookings", includeZero: true },
                            axisX: { title: "Month" },
                            data: [{ type: "column", dataPoints: response.dataPoints }]
                        });
                        chart.render();

        // Load FullCalendar
        function loadCalendar() {
            var calendarEl = document.getElementById('calendar');
            var calendar = new FullCalendar.Calendar(calendarEl, {
                initialView: 'dayGridMonth',
                headerToolbar: {
                    left: 'prev,next today',
                    center: 'title',
                    right: 'dayGridMonth,timeGridWeek,timeGridDay'
                },
                events: function(fetchInfo, successCallback, failureCallback) {
                    $.ajax({
                        url: 'admin_panel.php',
                        type: 'POST',
                        data: { action: 'get_calendar_bookings' },
                        dataType: 'json',
                        success: function(response) {
                            if (response.success) {
                                successCallback(response.events);
                            } else {
                                console.error('Calendar Bookings Error:', response.error);
                                failureCallback(new Error(response.error));
                            }
                        },
                        error: function(xhr, status, error) {
                            console.error('Calendar Bookings AJAX Error:', xhr.responseText);
                            failureCallback(new Error('Failed to load calendar bookings'));
                        }
                    });
                },
                eventContent: function(arg) {
                    return { html: '<div>' + arg.event.title + '</div>' };
                },
                eventMouseEnter: function(info) {
                    $(info.el).tooltip({
                        title: 'Hall: ' + info.event.extendedProps.hall_name + '<br>Time Slot: ' + info.event.extendedProps.time_slot,
                        html: true,
                        placement: 'top',
                        trigger: 'hover',
                        container: 'body'
                    });
                },
                dayCellDidMount: function(arg) {
                    var events = arg.el.querySelectorAll('.fc-event');
                    if (events.length > 0) {
                        arg.el.classList.add('fc-day-booked');
                    }
                }
            });
            calendar.render();
        }

        // Update total for Edit Booking form
        function updateEditTotal() {
            let hallPrice = parseInt($('#editHall').data('price')) || 0;
            let plates = parseInt($('#editPlates').val()) || 0;
            let cateringTotal = 0;
            let selectedItems = [];
            $('#editCateringMenu .menu-item:checked').each(function() {
                let price = parseInt($(this).data('price')) || 0;
                let name = $(this).data('name');
                cateringTotal += price * plates;
                selectedItems.push(name + " (₹" + price + ")");
            });
            let decorationPrice = parseInt($('#editDecoration option:selected').data('price')) || 0;
            let total = hallPrice + cateringTotal + decorationPrice;
            let hallName = $('#editHall').val() || "None";
            let decoration = $('#editDecoration').val() || "None";
            let discount = 0;
            if (['Oasis Lawn', 'Celestial Luxe'].includes(hallName)) {
                if (decoration === 'Premium') discount = 0.05;
                else if (decoration === 'Luxury') discount = 0.10;
            }
            let finalTotal = discount > 0 ? total * (1 - discount) : total;
            $('#editTotal').text(Math.round(finalTotal));
            $('#editSummaryText').html(
                "Hall: " + hallName + " (₹" + hallPrice + ")<br>" +
                "Catering: " + (selectedItems.length > 0 ? selectedItems.join(", ") : "None") +
                " × " + plates + " plates = ₹" + cateringTotal + "<br>" +
                "Decoration: " + decoration + " (₹" + decorationPrice + ")" +
                (discount > 0 ? "<br>Discount: " + (discount * 100) + "%" : "")
            );
        }

                let bookingId = $(this).data('id');

                    alert('Error: Invalid booking ID');
                    return;
                }
                $.ajax({
                    url: 'admin_panel.php',
                    type: 'POST',
                    data: { action: 'get_booking', id: bookingId },
                    dataType: 'json',
                    success: function(response) {
                        if (response.success && response.booking) {
                            let booking = response.booking;
                            $('#editBookingId').val(booking.id);
                            $('#editContact').val(booking.contact || '');
                            $('#editBookingDate').val(booking.booking_date || '');
                            $('#editTimeSlot').val(booking.time_slot || '');
                            $('#editPlates').val(booking.plates || 0);
                            $('#editDecoration').val(booking.decoration || '');

                            $('#editHall').val(booking.hall_name || '');
                            $('#editBookingForm .hall-card').removeClass('selected');
                            let hallCard = $('#editBookingForm .hall-card[data-name="' + booking.hall_name + '"]');
                            if (hallCard.length) {
                                hallCard.addClass('selected');
                                $('#editHall').data('price', hallCard.data('price') || 0);
                            } else {
                                $('#editHall').data('price', 0);
                            }

                            $('#editCateringMenu .menu-item').prop('checked', false);
                            let cateringItems = booking.catering_item && booking.catering_item.trim() !== '' ? booking.catering_item.split(', ') : [];
                            cateringItems.forEach(function(item) {
                                let checkbox = $('#editCateringMenu .menu-item[data-name="' + item + '"]');
                                if (checkbox.length) {
                                    checkbox.prop('checked', true);
                                }
                            });

                            updateEditTotal();
                            $('#editBookingModal').modal('show');
                        } else {
                            console.error('Get Booking Error:', response.error);
                            alert('Error: ' + response.error);
                        }
                    },
                    error: function(xhr, status, error) {
                        console.error('Get Booking AJAX Error:', xhr.responseText);
                        alert('An error occurred while fetching booking details.');
                    }
                });
            });
